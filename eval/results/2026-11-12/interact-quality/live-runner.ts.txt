import { appendFileSync, mkdtempSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import type { Database } from "bun:sqlite";
import { registerBrowserEffects } from "./api/application/browser/browser-effects";
import { createInteractInference } from "./api/application/browser/interact-inference";
import { searchThenDelegate } from "./api/application/toolchain/conversation-script.fixture";
import { harness } from "./api/application/toolchain/toolchain.fixture";
import {
	createBrowser,
	createBrowserInteract,
	type InteractInference,
} from "./api/domains/browser";
import {
	FakeEncoder,
	FakeFiles,
	fakeResolver,
} from "./api/domains/browser/test/fakes";
import { InteractFakeDriver } from "./api/domains/browser/test/interact.fixture";
import { sessionEvents } from "./api/domains/dialogue/repository/specialist";
import {
	createCloudAgents,
	type CloudAgents,
} from "./api/domains/inference/adapters/cloud-agents";
import { createCodexHost } from "./api/domains/inference/adapters/codex-host";
import { systemContext } from "./api/infrastructure/system-context";
import { waitFor } from "./api/infrastructure/wait-for";

const output = Bun.argv[2];
const modeFilter = Bun.argv[3];
if (modeFilter && !["baseline", "draft"].includes(modeFilter))
	throw Error("invalid_mode");
if (!output) throw Error("usage: bun interact-live.ts RESULTS.jsonl");
const write = (row: unknown) =>
	appendFileSync(output, JSON.stringify(row) + "\n");
const model = "gpt-6-luna",
	reasoningEffort = "low",
	budget = 18;
const cases = [
	{
		id: "rates",
		page: "入場料金表\n東館 一般 1,200円 夜間 800円\n西館 一般 1,500円 夜間 950円\n別館 一般 600円 夜間 400円",
		opening:
			"https://example.com/rates の料金表で、東館と西館の一般料金を施設名と金額だけで答える作業案を一緒に作ってください。今は準備までで、ページを開くのは私が確認してからにしてください。東館と西館を取り違えると見積りを出し直すので、順序は東館、西館でお願いします。",
		correction:
			"一般料金ではなく、両方とも夜間料金に訂正します。対象と順序、回答形式は同じです。",
		expected: ["東館", "800", "西館", "950"],
	},
	{
		id: "hours",
		page: "受付時間\n配送窓口 火曜 10:00〜16:30 水曜 11:00〜17:45\n受取窓口 火曜 09:00〜15:15 水曜 09:30〜16:10\n電話窓口 水曜 08:00〜18:30",
		opening:
			"https://example.com/hours で、配送窓口と受取窓口の火曜の受付終了時刻を調べたいです。まず作業案を出して、私の確認後にページを見てください。答えは配送、受取の順で、窓口名と終了時刻だけにしてください。開始時刻や電話窓口は含めないでください。",
		correction:
			"曜日だけ水曜に訂正します。終了時刻だけ、配送、受取の順はそのままです。",
		expected: ["配送", "17:45", "受取", "16:10"],
	},
] as const;
const sdkHome = mkdtempSync(join(tmpdir(), "interact-luna-"));
const codexHost = createCodexHost(process.env, { home: sdkHome });
if (!codexHost) throw Error("codex_cli_unavailable");
const cloud = createCloudAgents(process.env, { codexHost });
type Call = Parameters<CloudAgents["run"]>[0];
type Harness = Awaited<ReturnType<typeof harness>>;
const skill = systemContext("browser.contexts.interact.skill", {});
const reviewInstructions = systemContext(
	"browser.contexts.interact.review",
	{},
);
const confirmation =
	"その作業案で始めてください。ページを開いて、確認できた結果を答えてください。";

async function trial(task: (typeof cases)[number], mode: "baseline" | "draft") {
	const started = performance.now(),
		elapsed = () => Math.round(performance.now() - started);
	const usage: {
		seq: number;
		phase: string;
		model?: string | null;
		input?: number | null;
		output?: number | null;
		elapsedMs: number;
		value?: unknown;
		error?: string;
	}[] = [];
	const reviews: {
		revision: number;
		prompt: unknown;
		startMs: number;
		result?: unknown;
	}[] = [];
	const browser: {
		purpose: string;
		startMs: number;
		brief?: string;
		reviewInput?: unknown;
	}[] = [];
	let correctionSent = false;
	let calls = 0;
	const releaseOld = Promise.withResolvers<void>();
	async function complete(
		phase: string,
		messages: Call["messages"],
		signal?: AbortSignal,
		schema?: Call["outputSchema"],
	) {
		if (calls >= budget) throw Error("trial_budget_exhausted");
		const seq = ++calls,
			start = elapsed();
		try {
			const result = await cloud.run({
				kind: "codex-sdk",
				model,
				reasoningEffort,
				credential: null,
				baseUrl: "https://api.openai.com/v1",
				messages,
				...(schema ? { outputSchema: schema } : {}),
				signal: AbortSignal.any([
					AbortSignal.timeout(75_000),
					...(signal ? [signal] : []),
				]),
			});
			let value: unknown = result.text;
			try {
				value = JSON.parse(result.text);
			} catch {}
			usage.push({
				seq,
				phase,
				model: result.usageModel,
				input: result.inputTokens,
				output: result.outputTokens,
				elapsedMs: elapsed() - start,
				value,
			});
			return result.text;
		} catch (e) {
			usage.push({
				seq,
				phase,
				elapsedMs: elapsed() - start,
				error: e instanceof Error ? e.message : String(e),
			});
			throw e;
		}
	}

	const driver = new (class extends InteractFakeDriver {
		override async readText() {
			return {
				url: this.tabs[0]!.url,
				title: "試験ページ",
				text: task.page,
				charCount: task.page.length,
				truncated: false,
				root: "main" as const,
			};
		}
	})();
	const files = new FakeFiles();
	let infer!: InteractInference;
	let service!: ReturnType<typeof createBrowserInteract>;
	const h: Harness = await harness({
		pipelines: true,
		browser(store, queue) {
			const legacy = createBrowser({
				store,
				queue,
				driver: new InteractFakeDriver(),
				files: new FakeFiles(),
				encoder: new FakeEncoder(),
				webOrigins: [],
				resolveHost: fakeResolver(),
			});
			let guard: (db: Database, source: unknown) => boolean = () => false;
			service = createBrowserInteract({
				store,
				queue,
				now: Date.now,
				createDriver: () => driver,
				files,
				encoder: new FakeEncoder(),
				webOrigins: [],
				resolveHost: fakeResolver(),
				skill: {
					ref: "skill:browser.interact.operator@1",
					digest: "sha256:" + "a".repeat(64),
				},
				reviewInstructions,
				validSource: (db, ref) =>
					guard(db, (ref as { source: unknown }).source),
				infer(input, signal, current) {
					const messages = input.prepared.messages
						.map((m) => m.content)
						.join("\n");
					const reviewInput =
						input.purpose === "completion_review"
							? JSON.parse(input.prepared.messages.at(-1)!.content)
							: undefined;
					browser.push({
						purpose: input.purpose,
						startMs: elapsed(),
						brief: /"handedOverRequest":"((?:[^"\\]|\\.)*)"/.exec(
							messages,
						)?.[1],
						reviewInput,
					});
					return infer(input, signal, current);
				},
			});
			return {
				service: legacy,
				effects: registerBrowserEffects({ queue, service: legacy }),
				requests: { userMessageInTransaction: () => null },
				interact: {
					...service,
					setCollaborationSourceGuard: (next) => {
						guard = next;
					},
				},
			};
		},
		discoveryTarget: () => "collaborate_interact",
		firstRound: (_tools, _question, messages) =>
			searchThenDelegate(
				(c) => c.id === "collaboration.interact",
				task.opening,
				{ announcement: "ブラウザの共同作業を始めます。", impact: "create" },
			)(messages),
		async specialist(messages, signal, options) {
			const phase = messages.at(-1)?.content.startsWith("HOST_EVENT_NOTICE")
				? "notice"
				: "specialist";
			return complete(phase, messages, signal, options?.jsonSchema);
		},
		async interactControl(messages, signal, options) {
			const phase = messages[0]?.content.includes("完了を照合")
				? "completion"
				: "operator";
			return complete(phase, messages, signal, options?.jsonSchema);
		},
		async workflowReview(messages, signal, options) {
			const prompt = JSON.parse(messages.at(-1)!.content);
			const review: (typeof reviews)[number] = {
				revision: Number(prompt.target.version),
				prompt,
				startMs: elapsed(),
			};
			reviews.push(review);
			const hold = review.revision === 1 && !correctionSent;
			review.result = JSON.parse(
				await complete(
					"quick",
					messages,
					hold ? undefined : signal,
					options?.jsonSchema,
				),
			);
			if (hold) await releaseOld.promise;
			return review.result;
		},
	});
	infer = createInteractInference(
		{ store: h.store, inference: h.inference },
		skill,
	);
	const turns: Record<string, unknown>[] = [];
	const snapshots: unknown[] = [];
	const userMessages = () => h.conversation.get("main").messages;
	const read = () => {
		const session = h.dialogue.specialistSession("main");
		if (!session) return null;
		return h.store.read((db) => ({
			session,
			work: h.toolchain.collaboration.works.view(
				db,
				"interact",
				"main",
				session.workRef,
			),
			tools: sessionEvents(db, session.id).filter(
				(event) => event.kind === "tool",
			),
			notices: db
				.query(
					`SELECT o.event_json, r.id, r.status, r.answer_message_id
			FROM dialogue_specialist_observations o JOIN dialogue_runs r ON r.id=o.run_id
			WHERE r.specialist_session_id=? ORDER BY r.seq`,
				)
				.all(session.id),
		}));
	};
	async function send(phase: string, text: string) {
		const startMs = elapsed();
		const response = await h.request("/api/runs", {
			requestId: crypto.randomUUID(),
			conversationId: "main",
			text,
		});
		if (response.status !== 202) throw Error("http_" + response.status);
		const { id } = await response.json();
		const run = await h.dialogue.waitForTerminal(id, { timeoutMs: 100_000 });
		turns.push({
			phase,
			id,
			startMs,
			endMs: elapsed(),
			status: run?.status,
			error: run?.error,
			answer: h.dialogue.answerText(id),
		});
		snapshots.push(read());
	}
	const idle = () =>
		waitFor(() => !h.dialogue.specialistSession("main")?.activeRunId, {
			timeout: 25_000,
			interval: 50,
		});
	let error: string | undefined;
	try {
		const settings = h.settings.get();
		settings.routes.llm.mode = "larm-preferred";
		await h.settings.apply({
			requestId: crypto.randomUUID(),
			expectedRevision: settings.revision,
			settings,
			keys: [],
		});
		h.toolchain.registry.get(
			"adapter:browser.interact@1",
		)!.availableInTransaction = () => true;
		if (mode === "baseline")
			await h.store.write((db) =>
				h.toolchain.capabilities.publishToolchainRevisionInTransaction(
					db,
					"collaboration.interact",
					"toolchain:collaboration.interact@2",
				),
			);
		await Bun.sleep(0);
		await send("opening", task.opening);
		if (mode === "draft" && !reviews.length)
			await waitFor(() => reviews.length > 0, {
				timeout: 1500,
				interval: 50,
			}).catch(() => undefined);
		correctionSent = true;
		try {
			await send("correction", task.correction);
		} finally {
			releaseOld.resolve();
		}
		await idle();
		const beforeConfirmation = read()?.work?.revision;
		await send("confirmation", confirmation);
		const updatedInConfirmation = read()?.work;
		if (
			mode === "draft" &&
			updatedInConfirmation &&
			beforeConfirmation !== undefined &&
			updatedInConfirmation.revision > beforeConfirmation &&
			!updatedInConfirmation.current.child &&
			read()?.tools.some(
				(event) =>
					event.kind === "tool" &&
					event.error === "collaboration_confirmation_required",
			)
		) {
			await idle();
			await send("confirm-revised-draft", confirmation);
		}
		if (
			mode === "draft" &&
			read()?.tools.some(
				(event) =>
					event.kind === "tool" &&
					event.error === "collaboration_draft_conditions_changed",
			)
		) {
			const before = read()?.work?.revision;
			await send(
				"repair-draft",
				"訂正した条件で作業案を保存し直してください。ページは次の確認後に開いてください。",
			);
			await idle();
			const updated = read()?.work;
			if (
				updated &&
				before !== undefined &&
				updated.revision > before &&
				!updated.current.child
			)
				await send("confirm-revised-draft", confirmation);
		}
		const pending = () => {
			const work = read()?.work;
			const child = work?.current.child;
			return Boolean(
				work?.busy ||
				work?.current.review?.status === "running" ||
				child?.status === "running",
			);
		};
		await waitFor(() => !pending(), { timeout: 110_000, interval: 100 });
		const child = read()?.work?.current.child;
		if (child)
			await waitFor(
				() =>
					read()?.notices.some((row) => {
						const notice = row as {
							event_json: string;
							status: string;
							answer_message_id: string | null;
						};
						const event = JSON.parse(notice.event_json);
						return (
							event.id === child.id &&
							event.revision === child.revision &&
							notice.status === "completed" &&
							!!notice.answer_message_id
						);
					}) ?? false,
				{ timeout: 30_000, interval: 100 },
			).catch(() => undefined);
		await idle();
	} catch (e) {
		error = e instanceof Error ? e.message : String(e);
	} finally {
		releaseOld.resolve();
		write({
			id: task.id,
			mode,
			elapsedMs: elapsed(),
			turns,
			snapshots,
			browser,
			reviews,
			usage,
			final: read(),
			driverCalls: driver.calls,
			messages: userMessages(),
			error,
		});
		console.log(
			JSON.stringify({
				id: task.id,
				mode,
				elapsedMs: elapsed(),
				calls,
				reviews: reviews.length,
				browser: browser.length,
				error,
			}),
		);
		await service.close();
		await h.close();
	}
}
write({
	kind: "configuration",
	model,
	reasoningEffort,
	budget,
	cases,
	confirmation,
	route:
		"Codex subscription, fixed parent routing and isolated synthetic pages; native specialist, Quick, operator and completion inference",
});
try {
	await Promise.all(
		cases.map(async (task) => {
			if (modeFilter !== "draft") await trial(task, "baseline");
			if (modeFilter !== "baseline") await trial(task, "draft");
		}),
	);
	write({ kind: "end" });
} finally {
	await cloud.close?.();
	rmSync(sdkHome, { recursive: true, force: true });
}
